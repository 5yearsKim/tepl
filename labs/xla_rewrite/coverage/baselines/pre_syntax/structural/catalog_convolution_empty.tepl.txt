from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: convolution-empty
// Scope: General convolution; result shape is ordinary source/adapter metadata.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_convolution_empty {
    dtype T; X: T[Left...] W: T[Right...]
     (hlo.general_convolution[@c] X W) => (hlo.fill[@zero])
     where { product(Left)==0 || product(Right)==0 }
     derive { @zero={shape:@c.shape,value:0,element_type:T}; }
}
