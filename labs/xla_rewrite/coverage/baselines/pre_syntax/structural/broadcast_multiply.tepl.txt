from "dialects/hlo_structural.tepl" import HLO as hlo;
use hlo::{add, mul, negate, subtract, zero_like, transpose, broadcast,
        reduce_sum, reverse, slice, dot};

rule broadcast_multiply {
    (mul (broadcast[@left] X) (broadcast[@right] Y))
    => (broadcast[@left] (mul X Y))
    where { @left.dimensions == @right.dimensions
         && @left.new_extents == @right.new_extents }
}
