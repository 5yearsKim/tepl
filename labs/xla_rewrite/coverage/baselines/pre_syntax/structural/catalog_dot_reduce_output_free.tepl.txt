from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: dot-reduce-output-free
// Scope: Reduce the left free axis of a rank-two dot; implicit sum initializer zero.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_dot_reduce_output_free {
    X: [M,K] Y: [K,N]
     (hlo.reduce_sum[@r] (hlo.dot[@d] X Y)) => (hlo.dot[@out] (hlo.reduce_sum[@r] X) Y)
     where { @r.dimensions==[0] && @d.lhs_batching_dimensions==[] && @d.rhs_batching_dimensions==[] && @d.lhs_contracting_dimensions==[1] && @d.rhs_contracting_dimensions==[0] }
     derive { @out={lhs_batching_dimensions:[],rhs_batching_dimensions:[],lhs_contracting_dimensions:[0],rhs_contracting_dimensions:[0]}; }
}
