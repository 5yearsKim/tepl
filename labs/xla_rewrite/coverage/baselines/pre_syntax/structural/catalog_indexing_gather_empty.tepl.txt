from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: indexing-gather-empty
// Scope: Empty input; gather result shape is adapter metadata.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_gather_empty {
    dtype T; X: T[Dims...]
     (hlo.gather[@g] X Indices) => (hlo.fill[@zero])
     where { product(Dims)==0 }
     derive { @zero={shape:@g.shape,value:0,element_type:T}; }
}
