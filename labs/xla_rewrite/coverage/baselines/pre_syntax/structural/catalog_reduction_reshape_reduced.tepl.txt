from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: reduction-reshape-reduced
// Scope: Sum over a merged pair of reduced axes; rank-three input.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_reduction_reshape_reduced {
    X: [A,B,C]
     (hlo.reduce_sum[@r] (hlo.reshape[@s] X)) => (hlo.reduce_sum[@out] X)
     where { @s.shape==[A*B,C] && @r.dimensions==[0] }
     derive { @out={dimensions:[0,1]}; }
}
