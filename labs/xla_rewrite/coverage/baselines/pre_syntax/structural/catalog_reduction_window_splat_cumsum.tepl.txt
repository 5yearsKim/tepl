from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: reduction-window-splat-cumsum
// Scope: Rank-one cumulative sum of a scalar f32 broadcast; scalar constant can be a capture.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_reduction_window_splat_cumsum {
    C: f32[]
     (hlo.reduce_window_sum[@w] (hlo.broadcast[@b] C) 0.0:f32)
     => (hlo.multiply (hlo.broadcast[@b] C) (hlo.add (hlo.iota[@i]) (hlo.broadcast[@b] 1.0:f32)))
     where { @b.dimensions==[] && len(@b.new_extents)==1 && @w.window_dimensions==@b.new_extents && @w.window_strides==[1] && @w.base_dilations==[1] && @w.window_dilations==[1] && @w.padding_low==[@b.new_extents[0]-1] && @w.padding_high==[0] }
     derive { @i={shape:@b.new_extents,dimension:0,element_type:f32}; }
}
