from "dialects/hlo_structural.tepl" import HLO as hlo;
use hlo::{reduce_sum};

rule reduce_compose_leading {
    X: [I, J, K]
    (reduce_sum[@outer] (reduce_sum[@inner] X)) => (reduce_sum[@out] X)
    where { @inner.dimensions == [0] && @outer.dimensions == [0] }
    derive { @out = { dimensions: [0, 1] }; }
}
