from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: indexing-transpose-scatter
// Scope: Rank-two operand, rank-three updates with leading scatter axis, no batching.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_transpose_scatter {
    (hlo.transpose[@p] (hlo.scatter[@s] X Indices Updates))
     => (hlo.scatter[@out] (hlo.transpose[@p] X) Indices (hlo.transpose[@up] Updates))
     where { @p.permutation==[1,0] && @s.input_batching_dims==[] && @s.scatter_indices_batching_dims==[] && @s.update_window_dims==[1,2] && @s.inserted_window_dims==[] && @s.scatter_dims_to_operand_dims==[0] && @s.index_vector_dim==1 }
     derive { @out={..@s,scatter_dims_to_operand_dims:[1]}; @up={permutation:[0,2,1]}; }
}
