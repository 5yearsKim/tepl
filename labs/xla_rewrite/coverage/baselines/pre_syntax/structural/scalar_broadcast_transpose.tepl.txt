from "dialects/hlo_structural.tepl" import HLO as hlo;
use hlo::{add, mul, negate, subtract, zero_like, transpose, broadcast,
        reduce_sum, reverse, slice, dot};

rule scalar_broadcast_transpose {
    (transpose[@p] (broadcast[@b] C)) => (broadcast[@out] C)
    where { @b.dimensions == [] && len(@b.new_extents) == 2
         && @p.permutation == [1, 0] }
    derive { @out = { ..@b, new_extents: gather(@b.new_extents, [1, 0]) }; }
}
