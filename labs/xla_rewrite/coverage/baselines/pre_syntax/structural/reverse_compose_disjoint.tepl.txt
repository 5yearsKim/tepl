from "dialects/hlo_structural.tepl" import HLO as hlo;
use hlo::{add, mul, negate, subtract, zero_like, transpose, broadcast,
        reduce_sum, reverse, slice, dot};

rule reverse_compose_disjoint {
    X: [I, J]
    (reverse[@outer] (reverse[@inner] X)) => (reverse[@out] X)
    where { @inner.dimensions == [0] && @outer.dimensions == [1] }
    derive { @out = { dimensions: [0, 1] }; }
}
