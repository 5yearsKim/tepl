from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: indexing-scatter-scalar-map
// Scope: Single operand/result, empty scalar index vector, whole-tensor combining region.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_scatter_scalar_map {
    X: [Dims...] Y: [Dims...] Indices: [Zero]
     (hlo.scatter[@s] X Indices Y) => (hlo.map[@body] X Y)
     where { Zero==0 && @s.index_vector_dim==0 && @s.update_window_dims==range(len(Dims)) && @s.inserted_window_dims==[] && @s.scatter_dims_to_operand_dims==[] && @s.input_batching_dims==[] && @s.scatter_indices_batching_dims==[] }
     derive { @body={body:@s.body}; }
}
