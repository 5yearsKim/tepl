from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: indexing-pad-negative
// Scope: Rank-one low crop -1 and high padding +2; signed pad metadata retained and crop bounds use nonnegative shape arithmetic.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_pad_negative {
    X: [N]
     (hlo.pad[@p] X C) => (hlo.slice[@crop] (hlo.pad[@out] X C))
     where { N>=1 && @p.low==[-1] && @p.high==[2] && @p.interior==[0] }
     derive { @out={low:[0],high:[2],interior:[0]}; @crop={starts:[1],sizes:[N+1],strides:[1]}; }
}
