from "dialects/hlo_structural.tepl" import HLO as hlo;

// Catalog: indexing-dynamic-slice-iota
// Scope: Rank-one, scalar-sized slice, i32 starts, optional scalar factor captured as Scale.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_dynamic_slice_iota {
    Start: i32[] Scale: []
     (hlo.dynamic_slice[@s] (hlo.multiply (hlo.iota[@i]) Scale) Start)
     => (hlo.reshape[@r] (hlo.multiply (hlo.convert[@c] (hlo.clamp 0:i32 Start (hlo.scalar[@limit]))) Scale))
     where { len(@i.shape)==1 && @i.shape[0]>0 && @i.dimension==0 && @s.sizes==[1] }
     derive { @limit={value:@i.shape[0]-1,element_type:i32,shape:[]}; @c={to:@i.element_type}; @r={shape:[1]}; }
}
