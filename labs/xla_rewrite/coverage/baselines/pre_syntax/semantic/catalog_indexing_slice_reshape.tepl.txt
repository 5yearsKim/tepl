from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: indexing-slice-reshape
// Scope: Rank-two input flattened to rank one; slice selects complete rows.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_slice_reshape {
    X: [Rows,Cols]
     (hlo.slice[@s] (hlo.reshape[@r] X)) => (hlo.reshape[@out] (hlo.slice[@inner] X))
     where { Cols>0 && @r.shape==[Rows*Cols] && @s.strides==[1] && @s.starts[0]%Cols==0 && @s.sizes[0]%Cols==0 }
     derive { @inner={starts:[floor_div(@s.starts[0],Cols),0],sizes:[floor_div(@s.sizes[0],Cols),Cols],strides:[1,1]}; @out={shape:@s.sizes}; }
}
