from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: reduction-window-pad
// Scope: Single-result window; nonnegative padding equals the initializer; reducer region forwarded.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_reduction_window_pad {
    X: [Dims...] Init: []
     (hlo.reduce_window[@w] (hlo.nonnegative_pad[@p] X Init) Init) => (hlo.reduce_window[@out] X Init)
     where { all([@p.interior[i]==0 || @w.base_dilations[i]==1 for i in range(len(Dims))]) }
     derive { @out={..@w,padding_low:[@w.padding_low[i]+@w.base_dilations[i]*@p.low[i] for i in range(len(Dims))],padding_high:[@w.padding_high[i]+@w.base_dilations[i]*@p.high[i] for i in range(len(Dims))],base_dilations:[if @p.interior[i]==0 then @w.base_dilations[i] else @p.interior[i]+1 for i in range(len(Dims))]}; }
}
