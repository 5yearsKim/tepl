from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: indexing-concat-splat-to-pad
// Scope: Rank-one low-edge scalar padding; high-edge form is symmetric.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_concat_splat_to_pad {
    X: [N] C: []
     (hlo.concatenate[@c] (hlo.broadcast[@b] C) X) => (hlo.nonnegative_pad[@p] X C)
     where { @c.dimension==0 && @b.dimensions==[] && len(@b.new_extents)==1 }
     derive { @p={low:@b.new_extents,high:[0],interior:[0]}; }
}
