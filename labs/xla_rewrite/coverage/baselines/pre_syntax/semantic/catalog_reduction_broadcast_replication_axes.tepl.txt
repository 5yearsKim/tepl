from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: reduction-broadcast-replication-axes
// Scope: Reduce all appended replication axes; zero-initialized sum, symbolic replication count.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_reduction_broadcast_replication_axes {
    dtype T; X: T[Dims...]
     (hlo.reduce_sum[@r] (hlo.broadcast[@b] X)) => (hlo.multiply X (hlo.broadcast[@scale] (hlo.scalar[@count])))
     where { @b.dimensions==range(len(Dims)) && @r.dimensions==[len(Dims)+i for i in range(len(@b.new_extents))] }
     derive { @count={value:product(@b.new_extents),element_type:T,shape:[]}; @scale={dimensions:[],new_extents:Dims}; }
}
