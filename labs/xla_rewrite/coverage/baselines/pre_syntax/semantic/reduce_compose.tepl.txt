from "dialects/hlo.tepl" import HLO as hlo;
use hlo::{add, mul, negate, subtract, zero_like, transpose, broadcast,
        reduce_sum, reverse, slice, dot};

rule reduce_compose {
    X: [I, J, K]
    (reduce_sum[@outer] (reduce_sum[@inner] X)) => (reduce_sum[@out] X)
    where { @inner.dimensions == [1] && @outer.dimensions == [1] }
    derive { @out = { dimensions: [1, 2] }; }
}
