from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: broadcast-iota
// Scope: Arbitrary-rank iota; broadcast appends axes.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_broadcast_iota {
    (hlo.broadcast[@b] (hlo.iota[@i])) => (hlo.iota[@out])
     where { @b.dimensions == range(len(@i.shape)) }
     derive { @out = { ..@i, shape: concat(@i.shape,@b.new_extents), dimension: @b.dimensions[@i.dimension] }; }
}
