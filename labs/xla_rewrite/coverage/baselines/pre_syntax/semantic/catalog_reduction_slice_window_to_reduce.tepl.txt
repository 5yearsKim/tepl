from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: reduction-slice-window-to-reduce
// Scope: Last element of a rank-one full cumulative window; f32 sum.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_reduction_slice_window_to_reduce {
    X: f32[N]
     (hlo.slice[@s] (hlo.reduce_window_sum[@w] X 0.0:f32)) => (hlo.reshape[@out] (hlo.reduce_sum[@r] X))
     where { N>0 && @w.window_dimensions==[N] && @w.window_strides==[1] && @w.base_dilations==[1] && @w.window_dilations==[1] && @w.padding_low==[N-1] && @w.padding_high==[0] && @s.starts==[N-1] && @s.sizes==[1] && @s.strides==[1] }
     derive { @r={dimensions:[0]}; @out={shape:[1]}; }
}
