from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: reduction-unit-axes
// Scope: Zero-initialized sum specialization of the generic-combiner schema.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_reduction_unit_axes {
    X: [Dims...]
     (hlo.reduce_sum[@r] X) => (hlo.reshape[@out] X)
     where { all([Dims[d]==1 for d in @r.dimensions]) }
     derive { @out={shape:gather(Dims,exclude(range(len(Dims)),@r.dimensions))}; }
}
