from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: reduction-splat-or
// Scope: Arbitrary tensor shapes under the catalog arithmetic/applicability assumptions.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_reduction_splat_or {
    X: [] Init: []
     (hlo.reduce_or[@r] (hlo.broadcast[@b] X) Init) => (hlo.logical_or X Init)
     where { @b.dimensions==[] && product(@b.new_extents)>0 && @r.dimensions==range(len(@b.new_extents)) }
}
