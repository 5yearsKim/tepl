from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: indexing-add-two-scatters-concat
// Scope: Rank-one point-scatter operands with [count,1] indices and no batching.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_add_two_scatters_concat {
    I: [N,One] J: [M,One] U: [N] V: [M]
     (hlo.add (hlo.scatter_add[@a] A I U) (hlo.scatter_add[@b] B J V))
     => (hlo.scatter_add[@a] (hlo.add A B) (hlo.concatenate[@cat] I J) (hlo.concatenate[@cat] U V))
     where { One==1 && @a.update_window_dims==[] && @b.update_window_dims==[] && @a.inserted_window_dims==[0] && @b.inserted_window_dims==[0] && @a.scatter_dims_to_operand_dims==[0] && @b.scatter_dims_to_operand_dims==[0] && @a.index_vector_dim==1 && @b.index_vector_dim==1 && @a.input_batching_dims==[] && @b.input_batching_dims==[] && @a.scatter_indices_batching_dims==[] && @b.scatter_indices_batching_dims==[] && !@a.indices_are_sorted && !@b.indices_are_sorted && !@a.unique_indices && !@b.unique_indices }
     derive { @cat={dimension:0}; }
}
