from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: indexing-pad-empty
// Scope: Arbitrary tensor shapes under the catalog arithmetic/applicability assumptions.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_indexing_pad_empty {
    X: [Dims...] C: []
     (hlo.nonnegative_pad[@p] X C) => (hlo.broadcast[@b] C)
     where { product(Dims)==0 }
     derive { @b={dimensions:[],new_extents:[Dims[i]+max(Dims[i]-1,0)*@p.interior[i]+@p.low[i]+@p.high[i] for i in range(len(Dims))]}; }
}
