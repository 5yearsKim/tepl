from "dialects/hlo.tepl" import HLO as hlo;

// Catalog: reduction-window-empty
// Scope: Single-output reduction window with nonnegative edge padding and empty input.
// Source arithmetic/applicability conditions are recorded in the manifest.
rule catalog_reduction_window_empty {
    X: [Dims...] Init: []
     (hlo.reduce_window[@w] X Init) => (hlo.broadcast[@b] Init)
     where { product(Dims)==0 }
     derive {
         let input = [if Dims[i]==0 then 0 else 1+(Dims[i]-1)*@w.base_dilations[i] for i in range(len(Dims))];
         let window = [(@w.window_dimensions[i]-1)*@w.window_dilations[i]+1 for i in range(len(Dims))];
         let padded = [@w.padding_low[i]+input[i]+@w.padding_high[i] for i in range(len(Dims))];
         @b={dimensions:[],new_extents:[if padded[i]<window[i] then 0 else floor_div(padded[i]-window[i],@w.window_strides[i])+1 for i in range(len(Dims))]};
     }
}
