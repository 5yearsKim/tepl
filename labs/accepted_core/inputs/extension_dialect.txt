dialect Views {
op add(lhs: tensor, rhs: tensor) -> tensor {
        dtype(a, b) {
            assert a == b;
            assert is_numeric(a);
            yield a;
        }
        axes {
            bind S { lhs: all_axes(lhs), rhs: all_axes(rhs) };
            yield S;
        }
        semantics {
            yield Add(lhs, rhs);
        }
    }
op multiply(lhs: tensor, rhs: tensor) -> tensor {
        alias: mul;
        dtype(a, b) {
            assert a == b;
            assert is_numeric(a);
            yield a;
        }
        axes {
            bind S { lhs: all_axes(lhs), rhs: all_axes(rhs) };
            yield S;
        }
        semantics {
            yield Product(lhs, rhs);
        }
    }
op negate(input: tensor) -> tensor {
        dtype(t) { assert is_numeric(t); yield t; }
        axes { bind S { input: all_axes(input) }; yield S; }
        semantics { yield Product(Const(-1), input); }
    }
op dot_general(lhs: tensor, rhs: tensor) -> tensor {
        
        alias: dot;
        dtype(a, b) {
            assert a == b;
            assert is_numeric(a);
            yield a;
        }
        attrs {
            
            lhs_contracting_dimensions: axis[];
            rhs_contracting_dimensions: axis[];
            lhs_batching_dimensions: axis[];
            rhs_batching_dimensions: axis[];
        }
        axes {
            let lb = attrs.lhs_batching_dimensions;
            let rb = attrs.rhs_batching_dimensions;
            let lc = attrs.lhs_contracting_dimensions;
            let rc = attrs.rhs_contracting_dimensions;

            let lf = ordered_complement(all_axes(lhs), lb ++ lc);
            let rf = ordered_complement(all_axes(rhs), rb ++ rc);

            bind B { lhs: lb, rhs: rb };
            bind K { lhs: lc, rhs: rc };
            bind I { lhs: lf };
            bind J { rhs: rf };

            
            
            yield B ++ I ++ J;
        }
        semantics {
            yield Reduce(K, Product(lhs, rhs));
        }

        
        
        
        
        
        
        
    }
op broadcast_in_dim(input: tensor) -> tensor {
        alias: broadcast;
        attrs {
            dimensions: axis[];
            new_extents: extent[];
        }
        dtype(t) { yield t; }
        axes {
            bind S { input: all_axes(input) };
            bind N = group(attrs.new_extents);
            yield place(S, attrs.dimensions, N);
        }
        semantics { yield input; }
    }
op transpose(input: tensor) -> tensor {
        attrs {
            permutation: axis[];
        }
        dtype(t) {
            yield t;
        }
        axes {
            bind S { input: all_axes(input) };

            assert is_valid_axis_list(attrs.permutation, rank(input));
            assert len(attrs.permutation) == rank(input);
            
            yield permute(S, attrs.permutation);
        }
        semantics {
            yield input;
        }
    }
op reverse(input: tensor) -> tensor {
        attrs { dimensions: axis[]; }
        dtype(t) { yield t; }
        axes {
            bind S { input: all_axes(input) };
            assert is_valid_axis_list(attrs.dimensions, rank(input));
            yield S;
        }
        semantics {
            yield Read(input, IndexReverse(Coords(S), attrs.dimensions));
        }
    }
op reduce_sum(input: tensor) -> tensor {
        attrs {
            dimensions: axis[];
        }
        dtype(t) {
            assert is_numeric(t);
            yield t;
        }
        axes {
            let reduced = attrs.dimensions;
            let retained = ordered_complement(all_axes(input), reduced);

            bind K { input: reduced };
            bind F { input: retained };

            yield F;
        }
        semantics {
            yield Reduce(K, input);
        }
    }
  op remove_unit_axes(input: tensor) -> tensor {
    attrs { dimensions: axis[]; }
    dtype(t) { yield t; }
    axes {
      assert is_valid_axis_list(attrs.dimensions, rank(input));
      let sizes = gather(shape(input), attrs.dimensions);
      assert is_positive_extents(sizes);
      assert extent_le(sizes, offset_extents(extent_sub(sizes, sizes), 1));
      let retained = ordered_complement(all_axes(input), attrs.dimensions);
      bind U { input: attrs.dimensions };
      bind F { input: retained };
      yield F;
    }
    semantics { yield Reduce(U, input); }
  }
}
